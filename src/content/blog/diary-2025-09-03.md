```cpp
int numberOfPairs(vector<vector<int>>& points) {

        sort(points.begin(),points.end(),[](const auto&a,const auto&b){

            if(a[0]==b[0]) return a[1]>b[1];

            else return a[0]<b[0];

        });

        int n = points.size();

        int ans = 0;

        for(int i = 0;i<n-1;i++){

            int a = points[i][1];

            int bt = INT_MIN;

            for(int j = i+1;j<n;j++){

                int b = points[j][1];

                if(b>a) continue;

                if(b>bt) ans++,bt = b;

            }

        }

        return ans;

    }
```
